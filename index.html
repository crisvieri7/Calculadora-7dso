function procesarCapturaStats(event) { const file = event.target.files[0]; if (!file) return; const loadingText = document.getElementById('loadingText'); loadingText.style.display = 'block'; Tesseract.recognize(file, 'eng+spa', { logger: m => console.log(m) }).then(({ data: { text } }) => { loadingText.style.display = 'none'; console.log("Texto leído por OCR:", text); // Para depurar en consola si lo necesitas const lineas = text.split('\n').filter(l => l.trim().length > 0); let statIndex = 1; lineas.forEach(linea => { if (statIndex > 4) return; // Buscamos cualquier secuencia de dígitos o números con puntos/comas const numsMatch = linea.match(/\d+([.,]\d+)?/g); if (numsMatch && numsMatch.length > 0) { let valorStr = numsMatch[0].replace(',', '.'); let valorNum = parseFloat(valorStr); // Corrección clave: si Tesseract leyó un número entero sin punto (ej: 1353 o 3376) if (!valorStr.includes('.') && valorNum > 99) { valorNum = valorNum / 100; } // Limitamos a máximo 2 decimales para que coincida con el juego valorNum = Math.round(valorNum * 100) / 100; const tipoDetectado = detectarTipoStat(linea); // Asignamos al campo Actual document.getElementById(`act${statIndex}`).value = valorNum; if (tipoDetectado) { document.getElementById(`s${statIndex}_act`).value = tipoDetectado; } statIndex++; } }); actualizarPorcentajes(); if (statIndex > 1) { alert("¡Captura analizada y corregida con éxito! Revisa los valores cargados."); } else { alert("No se pudieron extraer datos legibles. Intenta con una imagen más recortada o clara."); } }).catch(err => { loadingText.style.display = 'none'; console.error(err); alert("Error al procesar la imagen."); }); }
